// QUERY MEJORADA CON CRITERIOS DE DESEMPATE
// Esta query agrega las siguientes columnas adicionales para desempate:
// - JG_LOCAL: Victorias como local (team1)
// - JG_VISIT: Victorias como visitante (team2)
// - GF_LOCAL: Goles a favor como local
// - GF_VISIT: Goles a favor como visitante
//
// Criterios de desempate aplicados en ORDER BY (siguiendo reglas FIFA):
// 1. PUNTOS (si aplica sistema de puntos)
// 2. DIF (diferencia de goles)
// 3. PAF (goles a favor totales)
// 4. JG (partidos ganados)
// 5. JG_VISIT (victorias como visitante - criterio FIFA para equipos empatados)

$query = "SELECT t.id,
        t.category_id,
        t.group_id,
        t.name as team,
        count('x') as JJ,
        count(case when PAF > PEC then 1 end) JG,
        count(case when PAF < PEC then 1 end) JP,
        count(case when PAF = PEC then 1 end) JE,
        count(case when PPAF > PPEC then 1 end) JGP,
        count(case when PPAF < PPEC then 1 end) JPP,
        sum(score.PAF) as PAF,
        sum(score.PEC) as PEC,
        sum(score.PAF) - sum(score.PEC) as DIF,
        count(case when ES_LOCAL = 1 and PAF > PEC then 1 end) as JG_LOCAL,
        count(case when ES_LOCAL = 0 and PAF > PEC then 1 end) as JG_VISIT,
        sum(case when ES_LOCAL = 1 then PAF else 0 end) as GF_LOCAL,
        sum(case when ES_LOCAL = 0 then PAF else 0 end) as GF_VISIT
        $points, t.logo
        FROM
        (
            (SELECT
                    h.team1_id team,
                    h.score_team1 PAF,
                    h.score_team2 PEC,
                    h.score_penalties_team1 PPAF,
                    h.score_penalties_team2 PPEC,
                    1 as ES_LOCAL
                    FROM games h
                    INNER JOIN journeys j on j.id = h.journey_id
                    WHERE
                    h.status = 'Finalizado' and
                    h.is_deleted = 0 and
                    h.type = 'Temporada Regular' and
                    j.show_public ='Si' and
                    j.tournament_id = ". $tournament_id ."
            )
        UNION ALL
            ( SELECT
                    v.team2_id team,
                    v.score_team2 PEF,
                    v.score_team1 PEC,
                    v.score_penalties_team2 PPAF,
                    v.score_penalties_team1 PPEC,
                    0 as ES_LOCAL
                    FROM games v
                    INNER JOIN journeys j on j.id = v.journey_id
                    WHERE
                    v.status = 'Finalizado' and
                    v.is_deleted = 0 and
                    v.type = 'Temporada Regular' and
                    j.show_public ='Si' and
                    j.tournament_id = ". $tournament_id ."
            )
        ) AS score
        JOIN teams t on t.id = score.team
        WHERE
        t.status = 'Activo'
        GROUP BY team ";

    if ( strlen($points)==0) {
        $query .= 'ORDER BY JG desc, DIF DESC, PAF DESC, JP asc';
    }

    if (strlen($points)>0) {
        $query .= 'ORDER BY PUNTOS DESC, DIF DESC, PAF DESC, JG DESC, JG_VISIT DESC';
    }

    if ($limit) {
        $query .= ' LIMIT ' . $limit;
    }
