SELECT t.id, t.category_id, t.name as team,
                count('x') as JJ,
                count(case when PAF > PEC then 1 end) JG,
                count(case when PAF < PEC then 1 end) JP,
                count(case when PAF = PEC then 1 end) JE,
                count(case when PPAF > PPEC then 1 end) JGP,
                count(case when PPAF < PPEC then 1 end) JPP,
                sum(score.PAF) as PAF,
                sum(score.PEC) as PEC,
                sum(score.PAF) - sum(score.PEC) as DIF
                , sum( case when PAF > PEC then 3 else 0 end
 + case when PPAF > PPEC then 2 else 0 end
 + case when PPAF < PPEC then 1 else 0 end ) as PUNTOS, t.logo, t.id
                FROM
                (
                    (SELECT
                            h.team1_id team,
                            h.score_team1 PAF,
                            h.score_team2 PEC,
                            h.score_penalties_team1 PPAF,
                            h.score_penalties_team2 PPEC
                            FROM games h
                            INNER JOIN journeys j on j.id = h.journey_id
                            WHERE
                            h.status = 'Finalizado' and
                            h.is_deleted = 0 and
                            h.type = 'Temporada Regular' and
                            j.show_public ='Si' and
                            j.tournament_id = 9
                    )
                UNION ALL
                    ( SELECT
                            v.team2_id team,
                            v.score_team2 PEF,
                            v.score_team1 PEC,
                            v.score_penalties_team2 PPAF,
                            v.score_penalties_team1 PPEC
                            FROM games v
                            INNER JOIN journeys j on j.id = v.journey_id
                            WHERE
                            v.status = 'Finalizado' and
                            v.is_deleted = 0 and
                            v.type = 'Temporada Regular' and
                            j.show_public ='Si' and
                            j.tournament_id = 9
                    )
                ) AS score
                JOIN teams t on t.id = score.team
                WHERE
                t.status = 'Activo'
                and t.id = 25
                GROUP BY team ORDER BY PUNTOS DESC, DIF DESC

select g.id, g.game_day, g.game_time, j.name as jornada, g.status, t1.name as team1, g.score_team1, t2.name as team2, g.score_team2, g.notes
from games g
join teams t1 on t1.id = g.team1_id
join teams t2 on t2.id = g.team2_id
join journeys j on j.id = g.journey_id
WHERE
	g.team1_id = 554
    or g.team2_id = 554
 order by game_day;



select t1.name, score.score1, t2.name, score.team2, score.score2
from
(
    (
        select h.team1_id team1, h.team2_id team2, h.score_team1 score1, h.score_team2 score2
		from games h
		WHERE
		h.status = 'Finalizado'
		and h.is_deleted = 0
		and h.type = 'Temporada Regular'
    )
    UNION ALL
    (
        select v.team2_id team1, v.team1_id team2, v.score_team2 score1, v.score_team1 score2
		from games v
		WHERE
		v.status = 'Finalizado'
		and v.is_deleted = 0
		and v.type = 'Temporada Regular'
    )
) AS score
JOIN teams t1 on t1.id = score.team1
JOIN teams t2 on t2.id = score.team2
WHERE
t1.status = 'Activo'
and t2.status = 'Activo';
